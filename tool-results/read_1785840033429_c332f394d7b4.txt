   422→          const basePrice = comp.material && prices.countertopMaterial[comp.material as keyof typeof prices.countertopMaterial] ? prices.countertopMaterial[comp.material as keyof typeof prices.countertopMaterial] : prices.countertopMaterial.Granite
   423→          return sqft * basePrice

   427→        const brand = comp.brand as Brand
   428→        if (brand && prices.tandemDrawers[brand]) {
   429→          const qty = parseFloat(comp.quantity) || 0
   430→          return qty * prices.tandemDrawers[brand]
   431→        }
   432→        return 0
   433→      }
   434→
   435→      case 'dustbinBTD': {
   436→        const dustbinBrand = comp.brand as Brand
   437→        if (dustbinBrand && prices.dustbinBTD[dustbinBrand]) {

   440→        }
   441→        return 0
   442→      }
   443→
   444→      case 'bottlePullout': {

   446→        if (bottleBrand && prices.bottlePullout[bottleBrand]) {

   502→      case 'overheadLoft': {
   503→        const loftSqft = calculateSqft(comp.height, comp.width)
   504→        const loftType = comp.loftType as LoftType
   505→        const finish = comp.finish
   506→        if (loftType && prices.overheadLoft[loftType]) {
   507→          const basePrice = prices.overheadLoft[loftType]
   508→          const finishPrice = getEffectiveRate(finish, prices.overheadFinish)
   509→          return loftSqft * (basePrice + finishPrice)
   510→        }
   511→        return 0
   512→      }
   513→
   514→      case 'profileShutter': {
   515→        const profileQty = parseFloat(comp.quantity) || 0
   516→        const profilePrice = parseFloat(comp.price) || 0
   517→        return profileQty * profilePrice
   518→      }
   519→
   520→      case 'handles': {
   521→        const handleFeet = parseFloat(comp.runningFeet) || 0
   522→        const handlePrice = parseFloat(comp.handlePrice) || 0
   523→        return handleFeet * handlePrice
   524→      }
   525→
   526→      default:
   527→        return 0
   528→    }
   529→  }

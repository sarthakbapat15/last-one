    58→const calculateSqft = (height: string, width: string): number => {
    59→  const h = parseFloat(height) || 0
    60→  const w = parseFloat(width) || 0
    61→  return (h * w) / 92903

    76→function getKitchenItems(components: any, kitchenType: string): ExportItem[] {
    77→  const items: ExportItem[] = []
    78→
    79→  // Component 1: Ply Verticals or Structure/Countertop
    80→  const c1 = components.component1 || {}
    81→  if (kitchenType === 'Semi-Modular') {
    82→    const qty = parseFloat(c1.quantity) || 0
    83→    if (qty > 0) {
    84→      items.push({
    85→        label: 'Ply Verticals',
    86→        subLabel: 'CARCASE',
    87→        quantity: qty,
    88→        totalSqft: qty,
    89→        rate: PRICES.plyVerticals,
    90→        amount: qty * PRICES.plyVerticals
    91→      })
    92→    }
    93→  } else {
    94→    const sqft = calculateSqft(c1.height, c1.width)
    95→    const basePrice = c1.material === 'Quartz' ? 2000 : 1500
    96→    if (sqft > 0 && c1.height && c1.width) {
    97→      items.push({
    98→        label: 'Structure / Countertop',
    99→        subLabel: c1.material || 'Granite',
   100→        l: parseFloat(c1.height) || 0,
   101→        b: parseFloat(c1.width) || 0,
   102→        sqft,
   103→        quantity: 1,
   104→        totalSqft: sqft,
   105→        rate: basePrice,
   106→        amount: sqft * basePrice
   107→      })
   108→    }
   109→  }
   110→
   111→  // Tandem Drawers
   112→  const td = components.tandemDrawers || {}
   113→  if (td.brand && PRICES.tandemDrawers[td.brand as keyof typeof PRICES.tandemDrawers]) {
   114→    const qty = parseFloat(td.quantity) || 0
   115→    const price = PRICES.tandemDrawers[td.brand as keyof typeof PRICES.tandemDrawers]
   116→    if (qty > 0) {
   117→      items.push({
   118→        label: 'Tandem Drawers',
   119→        subLabel: td.brand,
   120→        quantity: qty,
   121→        totalSqft: qty,
   122→        rate: price,
   123→        amount: qty * price
   124→      })
   125→    }
   126→  }
   127→
   128→  // Dustbin + BTD
   129→  const db = components.dustbinBTD || {}
   130→  if (db.brand && PRICES.dustbinBTD[db.brand as keyof typeof PRICES.dustbinBTD]) {
   131→    const qty = parseFloat(db.quantity) || 0
   132→    const price = PRICES.dustbinBTD[db.brand as keyof typeof PRICES.dustbinBTD]
   133→    if (qty > 0) {
   134→      items.push({
   135→        label: 'Dustbin + BTD',
   136→        subLabel: db.brand,
   137→        quantity: qty,
   138→        totalSqft: qty,
   139→        rate: price,
   140→        amount: qty * price
   141→      })
   142→    }
   143→  }
   144→
   145→  // Bottle Pullout
   146→  const bp = components.bottlePullout || {}
   147→  if (bp.brand && PRICES.bottlePullout[bp.brand as keyof typeof PRICES.bottlePullout]) {
   148→    const qty = parseFloat(bp.quantity) || 0
   149→    const price = PRICES.bottlePullout[bp.brand as keyof typeof PRICES.bottlePullout]
   150→    if (qty > 0) {
   151→      items.push({
   152→        label: 'Bottle Pullout',
   153→        subLabel: bp.brand,
   154→        quantity: qty,
   155→        totalSqft: qty,
   156→        rate: price,
   157→        amount: qty * price
   158→      })
   159→    }
   160→  }
   161→
   162→  // Wicker Basket
   163→  const wb = components.wickerBasket || {}
   164→  if (wb.brand && PRICES.wickerBasket[wb.brand as keyof typeof PRICES.wickerBasket]) {
   165→    const qty = parseFloat(wb.quantity) || 0
   166→    const price = PRICES.wickerBasket[wb.brand as keyof typeof PRICES.wickerBasket]
   167→    if (qty > 0) {
   168→      items.push({
   169→        label: 'Wicker Baskets',
   170→        subLabel: wb.brand,
   171→        quantity: qty,
   172→        totalSqft: qty,
   173→        rate: price,
   174→        amount: qty * price
   175→      })
   176→    }
   177→  }
   178→
   179→  // Tall Unit
   180→  const tu = components.tallUnit || {}
   181→  if (tu.height && tu.width && tu.tallPantryFinish) {
   182→    const sqft = calculateSqft(tu.height, tu.width)
   183→    const rate = PRICES.tallPantryFinish[tu.tallPantryFinish as keyof typeof PRICES.tallPantryFinish] || 0
   184→    if (sqft > 0 && rate > 0) {
   185→      items.push({
   186→        label: 'Tall Unit',
   187→        subLabel: `Carcass ${tu.tallPantryFinish}`,
   188→        l: parseFloat(tu.height) || 0,
   189→        b: parseFloat(tu.width) || 0,
   190→        sqft,
   191→        quantity: 1,
   192→        totalSqft: sqft,
   193→        rate,
   194→        amount: sqft * rate
   195→      })
   196→    }
   197→  }
   198→
   199→  // Pantry Unit
   200→  const pu = components.pantryUnit || {}
   201→  if (pu.height && pu.width && pu.tallPantryFinish) {
   202→    const sqft = calculateSqft(pu.height, pu.width)
   203→    const finishRate = PRICES.tallPantryFinish[pu.tallPantryFinish as keyof typeof PRICES.tallPantryFinish] || 0
   204→    let amount = sqft * finishRate
   205→    const accType = pu.accessories as string
   206→    const accPrice = accType ? PRICES.pantryAccessories[accType as keyof typeof PRICES.pantryAccessories] : 0
   207→    amount += accPrice || 0
   208→    if (sqft > 0 && (finishRate > 0 || accPrice > 0)) {
   209→      items.push({
   210→        label: 'Pantry Unit',
   211→        subLabel: `Carcass ${pu.tallPantryFinish}`,
   212→        l: parseFloat(pu.height) || 0,
   213→        b: parseFloat(pu.width) || 0,
   214→        sqft,
   215→        quantity: 1,
   216→        totalSqft: sqft,
   217→        rate: finishRate,
   218→        amount
   219→      })
   220→      if (accPrice > 0) {
   221→        items.push({
   222→          label: '',
   223→          subLabel: accType,
   224→          quantity: 1,
   225→          totalSqft: 1,
   226→          rate: accPrice,
   227→          amount: accPrice
   228→        })
   229→      }
   230→    }
   231→  }
   232→
   233→  // Base Carcase
   234→  const bcc = components.baseCarcase || {}
   235→  if (bcc.height && bcc.width) {
   236→    const sqft = calculateSqft(bcc.height, bcc.width)
   237→    if (sqft > 0) {
   238→      items.push({
   239→        label: 'Base Carcase',
   240→        subLabel: 'CARCASE',
   241→        l: parseFloat(bcc.height) || 0,
   242→        b: parseFloat(bcc.width) || 0,
   243→        sqft,
   244→        quantity: 1,
   245→        totalSqft: sqft,
   246→        rate: PRICES.baseCarcase,
   247→        amount: sqft * PRICES.baseCarcase
   248→      })
   249→    }
   250→  }
   251→
   252→  // Overhead Cabinet
   253→  const ohc = components.overheadCabinet || {}
   254→  if (ohc.height && ohc.width && ohc.overheadCabinetFinish) {
   255→    const sqft = calculateSqft(ohc.height, ohc.width)
   256→    const ohcRate = PRICES.overheadCabinetFinish[ohc.overheadCabinetFinish as keyof typeof PRICES.overheadCabinetFinish] || 0
   257→    if (sqft > 0 && ohcRate > 0) {
   258→      items.push({
   259→        label: 'Overhead Cabinet',
   260→        subLabel: ohc.overheadCabinetFinish,
   261→        l: parseFloat(ohc.height) || 0,
   262→        b: parseFloat(ohc.width) || 0,
   263→        sqft,
   264→        quantity: 1,
   265→        totalSqft: sqft,
   266→        rate: ohcRate,
   267→        amount: sqft * ohcRate
   268→      })
   269→    }
   270→  }
   271→
   272→  // Overhead Loft
   273→  const ol = components.overheadLoft || {}
   274→  if (ol.height && ol.width && ol.loftType) {
   275→    const sqft = calculateSqft(ol.height, ol.width)
   276→    const basePrice = PRICES.overheadLoft[ol.loftType as keyof typeof PRICES.overheadLoft] || 0
   277→    const finishPrice = ol.finish ? (PRICES.overheadFinish[ol.finish as keyof typeof PRICES.overheadFinish] || 0) : 0
   278→    const totalRate = basePrice + finishPrice
   279→    if (sqft > 0 && totalRate > 0) {
   280→      items.push({
   281→        label: 'Overhead Loft',
   282→        subLabel: `Carcass ${ol.loftType}`,
   283→        l: parseFloat(ol.height) || 0,
   284→        b: parseFloat(ol.width) || 0,
   285→        sqft,
   286→        quantity: 1,
   287→        totalSqft: sqft,
   288→        rate: totalRate,
   289→        amount: sqft * totalRate
   290→      })
   291→    }
   292→  }
   293→
   294→  // Profile Shutter
   295→  const ps = components.profileShutter || {}
   296→  const psQty = parseFloat(ps.quantity) || 0
   297→  const psPrice = parseFloat(ps.price) || PRICES.profileShutter
   298→  if (psQty > 0) {
   299→    items.push({
   300→      label: 'Profile Shutter with Glass',
   301→      quantity: psQty,
   302→      totalSqft: psQty,
   303→      rate: psPrice,
   304→      amount: psQty * psPrice
   305→    })
   306→  }
   307→
   308→  // Handles
   309→  const hd = components.handles || {}
   310→  const handleFeet = parseFloat(hd.runningFeet) || 0
   311→  const handlePrice = parseFloat(hd.handlePrice) || 0
   312→  if (handleFeet > 0 && handlePrice > 0) {
   313→    items.push({
   314→      label: 'Handles',
   315→      subLabel: hd.handleType,
   316→      quantity: handleFeet,
   317→      totalSqft: handleFeet,
   318→      rate: handlePrice,
   319→      amount: handleFeet * handlePrice
   320→    })
   321→  }
   322→
   323→  return items
   324→}

   477→      amount: swcSqft * PRICES.sittingWithCushion
   478→    })
   479→  }
   480→
   481→  return items
   482→}

   557→    // Helper to write a section

   559→      if (items.length === 0) return

   101→        label: 'Ply Verticals',

   522→export async function POST(request: NextRequest) {
   523→  try {
   524→    const body = await request.json()
   525→    const validatedData = validateData(body)
   526→
   527→    const workbook = new Workbook()
   528→
   529→    // ========================================
   530→    // SHEET 1: Quotation (olive+HGL format)
   531→    // ========================================
   532→    const qs = workbook.addWorksheet('Quotation')
   533→
   534→    // Company Header — Logo centered at top (replaces text)
   535→    const logoPath = path.join(process.cwd(), 'upload', 'pioneer 2.jpg')
   536→    const lw = validatedData.logoSettings?.width || 350
   537→    const lh = validatedData.logoSettings?.height || 140
   538→    const lpos = validatedData.logoSettings?.position || 'center'
   539→    // Column positions: left=0.1, center=1.5, right=4 (approximate for 8-col sheet)
   540→    const colPositions: Record<string, number> = { left: 0.1, center: 1.5, right: 4.5 }
   541→    if (fs.existsSync(logoPath)) {
   542→      const logoBuffer = fs.readFileSync(logoPath)
   543→      const logoImageId = workbook.addImage({
   544→        buffer: logoBuffer,
   545→        extension: 'jpeg',
   546→      })
   547→      qs.addImage(logoImageId, {
   548→        tl: { col: colPositions[lpos] || 1.5, row: 0 },
   549→        ext: { width: lw, height: lh },
   550→      })
   551→    }
   552→
   553→    // Address below logo, centered
   554→    qs.mergeCells('A3:H3')
   555→    const c2 = qs.getCell('A3')

   483→        return 0
   484→      }
   485→
   486→      case 'wickerBasket': {

   488→        if (wickerBrand && prices.wickerBasket[wickerBrand]) {
   489→          const qty = parseFloat(comp.quantity) || 0
   490→          return qty * prices.wickerBasket[wickerBrand]
   491→        }
   492→        return 0
   493→      }
   494→
   495→      case 'tallUnit': {
   496→        const tallSqft = calculateSqft(comp.height, comp.width)
   497→        const depth = (comp.depth || '450mm') as TallUnitDepth
   498→        const tallFinish = comp.tallPantryFinish as TallUnitDepthFinishType
   499→        if (tallFinish === 'Postforming') return tallSqft * (parseFloat(postformingRate) || 0)
   500→        const depthPrices = prices.tallUnitFinishByDepth[depth]
   501→        const rate = depthPrices?.[tallFinish] || 0
   502→        return tallSqft * rate
   503→      }
   504→
   505→      case 'pantryUnit': {
   506→        const pantrySqft = calculateSqft(comp.height, comp.width)
   507→        const pantryFinish = comp.tallPantryFinish
   508→        let pantryTotal = 0
   509→        const pantryRate = getEffectiveRate(pantryFinish, prices.tallPantryFinish)

   514→        if (accessories && prices.pantryAccessories[accessories]) {
   515→          pantryTotal += prices.pantryAccessories[accessories]
   516→        }
   517→        return pantryTotal
   518→      }
   519→
   520→      case 'baseCarcase': {
   521→        const baseCarcaseSqft = calculateSqft(comp.height, comp.width)
   522→        return baseCarcaseSqft * prices.baseCarcase
   523→      }
   524→
   525→      case 'kitchenPaneling': {
   526→        const panelingSqft = calculateSqft(comp.height, comp.width)
   527→        const panelingFinish = comp.tallPantryFinish
   528→        const panelingRate = getEffectiveRate(panelingFinish, prices.kitchenPaneling)
   529→        return panelingSqft * panelingRate
   530→      }
   531→
   532→      case 'overheadCabinet': {
   533→        const overheadCabinetSqft = calculateSqft(comp.height, comp.width)
   534→        const ohcRate = getEffectiveRate(comp.overheadCabinetFinish, prices.overheadCabinetFinish)
   535→        return overheadCabinetSqft * ohcRate
   536→      }
   537→
   538→      case 'overheadLoft': {
   539→        const loftSqft = calculateSqft(comp.height, comp.width)
   540→        const loftType = comp.loftType as LoftType
   541→        const finish = comp.finish

   543→          const depth = (comp.depth || '450mm') as TallUnitDepth

   545→          const boxPrices = prices.overheadBoxLoftFinish[depth]

   554→        return 0
   555→      }
   556→
   557→      case 'profileShutter': {

   564→      case 'magicCorner': {

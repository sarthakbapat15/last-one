   259→          pantryTotal += prices.pantryAccessories[accessories]

   262→      }
   263→
   264→      case 'baseCarcase': {
   265→        const baseCarcaseSqft = calculateSqft(comp.height, comp.width)
   266→        return baseCarcaseSqft * prices.baseCarcase
   267→      }

   269→      case 'overheadCabinet': {
   270→        const overheadCabinetSqft = calculateSqft(comp.height, comp.width)

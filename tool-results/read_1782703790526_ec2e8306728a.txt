   137→    const h = parseFloat(height) || 0
   138→    const w = parseFloat(width) || 0
   139→    return (h * w) / 92903

   142→  const calculateComponentTotal = (component: keyof KitchenEstimate['components']): number => {
   143→    const comp = estimate.components[component]
   144→
   145→    switch (component) {
   146→      case 'component1':
   147→        if (kitchenType === 'Semi-Modular') {
   148→          const qty = parseFloat(comp.quantity) || 0
   149→          return qty * PRICES.plyVerticals
   150→        } else {
   151→          const sqft = calculateSqft(comp.height, comp.width)
   152→          const basePrice = comp.material === 'Quartz' ? 2000 : 1500
   153→          return sqft * basePrice
   154→        }
   155→
   156→      case 'tandemDrawers': {
   157→        const brand = comp.brand as Brand
   158→        if (brand && PRICES.tandemDrawers[brand]) {
   159→          const qty = parseFloat(comp.quantity) || 0
   160→          return qty * PRICES.tandemDrawers[brand]
   161→        }
   162→        return 0
   163→      }
   164→
   165→      case 'dustbinBTD': {
   166→        const dustbinBrand = comp.brand as Brand
   167→        if (dustbinBrand && PRICES.dustbinBTD[dustbinBrand]) {
   168→          const qty = parseFloat(comp.quantity) || 0
   169→          return qty * PRICES.dustbinBTD[dustbinBrand]
   170→        }
   171→        return 0
   172→      }
   173→
   174→      case 'bottlePullout': {
   175→        const bottleBrand = comp.brand as Brand
   176→        if (bottleBrand && PRICES.bottlePullout[bottleBrand]) {
   177→          const qty = parseFloat(comp.quantity) || 0
   178→          return qty * PRICES.bottlePullout[bottleBrand]
   179→        }
   180→        return 0
   181→      }
   182→
   183→      case 'wickerBasket': {
   184→        const wickerBrand = comp.brand as Brand
   185→        if (wickerBrand && PRICES.wickerBasket[wickerBrand]) {
   186→          const qty = parseFloat(comp.quantity) || 0
   187→          return qty * PRICES.wickerBasket[wickerBrand]
   188→        }
   189→        return 0
   190→      }
   191→
   192→      case 'tallUnit': {
   193→        const tallSqft = calculateSqft(comp.height, comp.width)
   194→        const tallFinish = comp.tallPantryFinish as TallPantryFinishType
   195→        if (tallFinish && PRICES.tallPantryFinish[tallFinish]) {
   196→          return tallSqft * PRICES.tallPantryFinish[tallFinish]
   197→        }
   198→        return 0
   199→      }
   200→
   201→      case 'pantryUnit': {
   202→        const pantrySqft = calculateSqft(comp.height, comp.width)
   203→        const pantryFinish = comp.tallPantryFinish as TallPantryFinishType

   205→        if (pantryFinish && PRICES.tallPantryFinish[pantryFinish]) {
   206→          pantryTotal = pantrySqft * PRICES.tallPantryFinish[pantryFinish]
   207→        }
   208→        const accessories = comp.accessories as AccessoriesType
   209→        if (accessories && PRICES.pantryAccessories[accessories]) {
   210→          pantryTotal += PRICES.pantryAccessories[accessories]
   211→        }
   212→        return pantryTotal
   213→      }
   214→
   215→      case 'baseCarcase': {
   216→        const baseCarcaseSqft = calculateSqft(comp.height, comp.width)
   217→        return baseCarcaseSqft * PRICES.baseCarcase
   218→      }
   219→
   220→      case 'overheadCabinet': {
   221→        const overheadCabinetSqft = calculateSqft(comp.height, comp.width)
   222→        const ohcFinish = comp.overheadCabinetFinish as OverheadCabinetFinishType
   223→        if (ohcFinish && PRICES.overheadCabinetFinish[ohcFinish]) {
   224→          return overheadCabinetSqft * PRICES.overheadCabinetFinish[ohcFinish]
   225→        }
   226→        return 0
   227→      }
   228→
   229→      case 'overheadLoft': {
   230→        const loftSqft = calculateSqft(comp.height, comp.width)
   231→        const loftType = comp.loftType as LoftType
   232→        const finish = comp.finish as FinishType
   233→        if (loftType && PRICES.overheadLoft[loftType]) {
   234→          const basePrice = PRICES.overheadLoft[loftType]

   236→          return loftSqft * (basePrice + finishPrice)

   238→        return 0
   239→      }
   240→

   242→        const profileQty = parseFloat(comp.quantity) || 0

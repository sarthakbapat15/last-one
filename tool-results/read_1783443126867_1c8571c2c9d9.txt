   512→  const calcBRSqft = (h: string, w: string) => {
   513→    const height = parseFloat(h) || 0
   514→    const width = parseFloat(w) || 0
   515→    return (height * width) / 92903
   516→  }

   518→  const calculateWardrobeTotal = (br: BedroomData): number => {
   519→    const sqft = calcBRSqft(br.wardrobe.height, br.wardrobe.width)
   520→    const rate = prices.bedroomWardrobeFinish[br.wardrobe.finish] || 0
   521→    let total = sqft * rate
   522→    if (br.wardrobe.slidingMechanism) total += prices.bedroomWardrobeSlidingMechanism
   523→    return total
   524→  }
   525→
   526→  const calculateBedroomLoftTotal = (br: BedroomData): number => {
   527→    const sqft = calcBRSqft(br.loft.height, br.loft.width)
   528→    const rate = br.loft.loftType && br.loft.finish
   529→      ? (prices.bedroomLoftFinish[br.loft.loftType]?.[br.loft.finish] || 0)
   530→      : 0
   531→    return sqft * rate
   532→  }
   533→
   534→  const calculateWindowSeatTotal = (br: BedroomData): number => {
   535→    const sqft = calcBRSqft(br.windowSeat.height, br.windowSeat.width)
   536→    return sqft * (prices.bedroomTallUnitFinish[br.windowSeat.finish as keyof typeof prices.bedroomTallUnitFinish] || 0)
   537→  }
   538→
   539→  const calculateStudyTableTotal = (br: BedroomData): number => {
   540→    const baseSqft = calcBRSqft(br.studyTable.base.height, br.studyTable.base.width)
   541→    const ohSqft = calcBRSqft(br.studyTable.overhead.height, br.studyTable.overhead.width)
   542→    return (baseSqft + ohSqft) * (prices.bedroomTallUnitFinish[br.studyTable.finish as keyof typeof prices.bedroomTallUnitFinish] || 0)
   543→  }
   544→
   545→  const calculateDresserUnitTotal = (br: BedroomData): number => {
   546→    const baseSqft = calcBRSqft(br.dresserUnit.baseDrawers.height, br.dresserUnit.baseDrawers.width)
   547→    const msSqft = calcBRSqft(br.dresserUnit.mirrorWithStorage.height, br.dresserUnit.mirrorWithStorage.width)
   548→    const mbpSqft = calcBRSqft(br.dresserUnit.mirrorOnBackPanel.height, br.dresserUnit.mirrorOnBackPanel.width)
   549→    return (baseSqft + msSqft + mbpSqft) * (prices.bedroomTallUnitFinish[br.dresserUnit.finish as keyof typeof prices.bedroomTallUnitFinish] || 0)
   550→  }
   551→
   552→  const calculateBedTotal = (br: BedroomData): number => {
   553→    if (br.bed.typeOfBed === 'Open Bed with Legs') return prices.bedroomOpenBedPrice
   554→    const sqft = calcBRSqft(br.bed.height, br.bed.width)
   555→    const rate = prices.bedroomTallUnitFinish[br.bed.finish as keyof typeof prices.bedroomTallUnitFinish] || 0
   556→    let total = sqft * rate
   557→    if (br.bed.typeOfBed === 'Hydraulic (Automatic)' || br.bed.typeOfBed === 'Pullout Trolly Bed') {
   558→      total += prices.bedroomHydraulicMechanismPrice
   559→    }
   560→    return total
   561→  }

   563→  const calculateHeadBoardTotal = (br: BedroomData): number => {
   564→    const sqft = calcBRSqft(br.headBoard.length, br.headBoard.width)
   565→    return sqft * (prices.bedroomHeadBoardRates[br.headBoard.headBoardType] || 0)
   566→  }

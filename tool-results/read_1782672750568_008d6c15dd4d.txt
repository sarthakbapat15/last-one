   320→   320→    if (sqft > 0 && rate > 0) {
   321→   321→      items.push({ label: 'Bedroom Loft', subLabel: `${loft.loftType} - ${loft.finish}`, l: parseFloat(loft.height) || 0, b: parseFloat(loft.width) || 0, sqft, quantity: 1, totalSqft: sqft, rate, amount: sqft * rate })
   322→   322→    }
   323→   323→  }
   324→   324→
   325→   325→  // Bed
   326→   326→  const bed = bedroomEstimate.bed || {}
   327→   327→  if (bed.length && bed.width && bed.finish) {
   328→   328→    const sqft = calculateSqft(bed.length, bed.width)
   329→   329→    const rate = PRICES.bedFinish[bed.finish as keyof typeof PRICES.bedFinish] || 0
   330→   330→    if (sqft > 0 && rate > 0) {
   331→   331→      const subLabel = `${bed.bedSize || ''} ${bed.storageType || ''} ${bed.finish}`.trim()
   332→   332→      items.push({ label: 'Bed', subLabel, l: parseFloat(bed.length) || 0, b: parseFloat(bed.width) || 0, sqft, quantity: 1, totalSqft: sqft, rate, amount: sqft * rate })
   333→   333→    }
   334→   334→  }
   335→   335→
   336→   336→  // Head Board
   337→   337→  const hb = bedroomEstimate.headBoard || {}
   338→   338→  if (hb.length && hb.width && hb.headBoardType) {
   339→   339→    const sqft = calculateSqft(hb.length, hb.width)
   340→   340→    const rate = PRICES.headBoardRates[hb.headBoardType as keyof typeof PRICES.headBoardRates] || 0
   341→   341→    if (sqft > 0 && rate > 0) {
   342→   342→      items.push({ label: 'Head Board', subLabel: hb.headBoardType, l: parseFloat(hb.length) || 0, b: parseFloat(hb.width) || 0, sqft, quantity: 1, totalSqft: sqft, rate, amount: sqft * rate })
   343→   343→    }
   344→   344→  }
   345→   345→
   346→   346→  return items
   347→   347→}

   351→   351→}

   218→  const calculateSqft = (height: string, width: string): number => {
   219→    const h = parseFloat(height) || 0
   220→    const w = parseFloat(width) || 0
   221→    return (h * w) / 92903

   224→  const calculateComponentTotal = (component: keyof KitchenEstimate['components']): number => {
   225→    const comp = estimate.components[component]
   226→
   227→    switch (component) {
   228→      case 'component1':
   229→        if (kitchenType === 'Semi-Modular') {
   230→          const qty = parseFloat(comp.quantity) || 0
   231→          return qty * PRICES.plyVerticals
   232→        } else {
   233→          const sqft = calculateSqft(comp.height, comp.width)
   234→          const basePrice = comp.material === 'Quartz' ? 2000 : 1500
   235→          return sqft * basePrice
   236→        }
   237→
   238→      case 'tandemDrawers': {
   239→        const brand = comp.brand as Brand
   240→        if (brand && PRICES.tandemDrawers[brand]) {
   241→          const qty = parseFloat(comp.quantity) || 0
   242→          return qty * PRICES.tandemDrawers[brand]
   243→        }
   244→        return 0
   245→      }
   246→
   247→      case 'dustbinBTD': {
   248→        const dustbinBrand = comp.brand as Brand
   249→        if (dustbinBrand && PRICES.dustbinBTD[dustbinBrand]) {
   250→          const qty = parseFloat(comp.quantity) || 0
   251→          return qty * PRICES.dustbinBTD[dustbinBrand]
   252→        }
   253→        return 0
   254→      }
   255→
   256→      case 'bottlePullout': {
   257→        const bottleBrand = comp.brand as Brand
   258→        if (bottleBrand && PRICES.bottlePullout[bottleBrand]) {
   259→          const qty = parseFloat(comp.quantity) || 0
   260→          return qty * PRICES.bottlePullout[bottleBrand]
   261→        }
   262→        return 0
   263→      }
   264→
   265→      case 'wickerBasket': {
   266→        const wickerBrand = comp.brand as Brand
   267→        if (wickerBrand && PRICES.wickerBasket[wickerBrand]) {
   268→          const qty = parseFloat(comp.quantity) || 0
   269→          return qty * PRICES.wickerBasket[wickerBrand]
   270→        }
   271→        return 0
   272→      }
   273→
   274→      case 'tallUnit': {
   275→        const tallSqft = calculateSqft(comp.height, comp.width)
   276→        const tallFinish = comp.tallPantryFinish as TallPantryFinishType

   278→          return tallSqft * PRICES.tallPantryFinish[tallFinish]
   279→        }
   280→        return 0
   281→      }
   282→
   283→      case 'pantryUnit': {
   284→        const pantrySqft = calculateSqft(comp.height, comp.width)
   285→        const pantryFinish = comp.tallPantryFinish as TallPantryFinishType

   288→          pantryTotal = pantrySqft * PRICES.tallPantryFinish[pantryFinish]
   289→        }
   290→        const accessories = comp.accessories as AccessoriesType
   291→        if (accessories && PRICES.pantryAccessories[accessories]) {
   292→          pantryTotal += PRICES.pantryAccessories[accessories]
   293→        }
   294→        return pantryTotal
   295→      }
   296→
   297→      case 'baseCarcase': {
   298→        const baseCarcaseSqft = calculateSqft(comp.height, comp.width)
   299→        return baseCarcaseSqft * PRICES.baseCarcase
   300→      }
   301→
   302→      case 'overheadCabinet': {
   303→        const overheadCabinetSqft = calculateSqft(comp.height, comp.width)
   304→        const ohcFinish = comp.overheadCabinetFinish as OverheadCabinetFinishType
   305→        if (ohcFinish && PRICES.overheadCabinetFinish[ohcFinish]) {
   306→          return overheadCabinetSqft * PRICES.overheadCabinetFinish[ohcFinish]
   307→        }
   308→        return 0
   309→      }
   310→
   311→      case 'overheadLoft': {

   316→          const basePrice = PRICES.overheadLoft[loftType]
   317→          const finishPrice = finish ? PRICES.overheadFinish[finish] : 0
   318→          return loftSqft * (basePrice + finishPrice)
   319→        }
   320→        return 0
   321→      }
   322→

   324→        const profileQty = parseFloat(comp.quantity) || 0
   325→        const profilePrice = parseFloat(comp.price) || 0
   326→        return profileQty * profilePrice
   327→      }

   329→      case 'handles': {
   330→        const handleFeet = parseFloat(comp.runningFeet) || 0
   331→        const handlePrice = parseFloat(comp.handlePrice) || 0

    77→const calculateSqft = (height: string, width: string): number => {
    78→  const h = parseFloat(height) || 0
    79→  const w = parseFloat(width) || 0
    80→  return (h * w) / 92903

    95→function getKitchenItems(components: any, kitchenType: string): ExportItem[] {
    96→  const items: ExportItem[] = []
    97→
    98→  // Component 1: Ply Verticals or Structure/Countertop
    99→  const c1 = components.component1 || {}
   100→  if (kitchenType === 'Semi-Modular') {
   101→    const qty = parseFloat(c1.quantity) || 0
   102→    if (qty > 0) {
   103→      items.push({
   104→        label: 'Ply Verticals',
   105→        subLabel: 'CARCASE',
   106→        quantity: qty,
   107→        totalSqft: qty,
   108→        rate: PRICES.plyVerticals,
   109→        amount: qty * PRICES.plyVerticals
   110→      })
   111→    }
   112→  } else {
   113→    const sqft = calculateSqft(c1.height, c1.width)
   114→    const basePrice = c1.material === 'Quartz' ? 2000 : 1500
   115→    if (sqft > 0 && c1.height && c1.width) {
   116→      items.push({
   117→        label: 'Structure / Countertop',
   118→        subLabel: c1.material || 'Granite',
   119→        l: parseFloat(c1.height) || 0,
   120→        b: parseFloat(c1.width) || 0,
   121→        sqft,
   122→        quantity: 1,
   123→        totalSqft: sqft,
   124→        rate: basePrice,
   125→        amount: sqft * basePrice
   126→      })
   127→    }
   128→  }
   129→
   130→  // Tandem Drawers
   131→  const td = components.tandemDrawers || {}
   132→  if (td.brand && PRICES.tandemDrawers[td.brand as keyof typeof PRICES.tandemDrawers]) {
   133→    const qty = parseFloat(td.quantity) || 0
   134→    const price = PRICES.tandemDrawers[td.brand as keyof typeof PRICES.tandemDrawers]

   137→        label: 'Tandem Drawers',

   180→

   271→  // Overhead Cabinet

   274→    const sqft = calculateSqft(ohc.height, ohc.width)
   275→    const ohcRate = PRICES.overheadCabinetFinish[ohc.overheadCabinetFinish as keyof typeof PRICES.overheadCabinetFinish] || 0
   276→    if (sqft > 0 && ohcRate > 0) {

   278→        label: 'Overhead Cabinet',
   279→        subLabel: ohc.overheadCabinetFinish,
   280→        l: parseFloat(ohc.height) || 0,
   281→        b: parseFloat(ohc.width) || 0,
   282→        sqft,
   283→        quantity: 1,
   284→        totalSqft: sqft,
   285→        rate: ohcRate,

   287→      })
   288→    }
   289→  }
   290→
   291→  // Overhead Loft
   292→  const ol = components.overheadLoft || {}
   293→  if (ol.height && ol.width && ol.loftType) {
   294→    const sqft = calculateSqft(ol.height, ol.width)
   295→    const basePrice = PRICES.overheadLoft[ol.loftType as keyof typeof PRICES.overheadLoft] || 0
   296→    const finishPrice = ol.finish ? (PRICES.overheadFinish[ol.finish as keyof typeof PRICES.overheadFinish] || 0) : 0
   297→    const totalRate = basePrice + finishPrice
   298→    if (sqft > 0 && totalRate > 0) {
   299→      items.push({
   300→        label: 'Overhead Loft',
   301→        subLabel: `Carcass ${ol.loftType}`,
   302→        l: parseFloat(ol.height) || 0,
   303→        b: parseFloat(ol.width) || 0,
   304→        sqft,
   305→        quantity: 1,
   306→        totalSqft: sqft,
   307→        rate: totalRate,
   308→        amount: sqft * totalRate
   309→      })
   310→    }
   311→  }
   312→
   313→  // Profile Shutter
   314→  const ps = components.profileShutter || {}
   315→  const psQty = parseFloat(ps.quantity) || 0
   316→  const psPrice = parseFloat(ps.price) || PRICES.profileShutter
   317→  if (psQty > 0) {
   318→    items.push({
   319→      label: 'Profile Shutter with Glass',
   320→      quantity: psQty,
   321→      totalSqft: psQty,

   329→  const handleFeet = parseFloat(hd.runningFeet) || 0

   336→      totalSqft: handleFeet,
   337→      rate: handlePrice,

   340→  }
   341→

   503→function getMiscItems(miscEstimate: any): ExportItem[] {
   504→  const items: ExportItem[] = []
   505→  if (!miscEstimate) return items
   506→
   507→  // False Ceiling
   508→  const fc = miscEstimate.falseCeiling || {}
   509→  if (fc.height && fc.width && fc.material) {
   510→    const sqft = (parseFloat(fc.height) * parseFloat(fc.width)) / 92903
   511→    const rate = PRICES.falseCeilingMaterial[fc.material as keyof typeof PRICES.falseCeilingMaterial] || 0
   512→    if (sqft > 0 && rate > 0) {
   513→      items.push({
   514→        label: 'False Ceiling',
   515→        subLabel: `${fc.ceilingType || ''} - ${fc.material}`,
   516→        l: parseFloat(fc.height) || 0,
   517→        b: parseFloat(fc.width) || 0,
   518→        sqft,
   519→        quantity: 1,
   520→        totalSqft: sqft,
   521→        rate,
   522→        amount: sqft * rate
   523→      })
   524→    }
   525→  }
   526→
   527→  // Electrical Work
   528→  const ew = miscEstimate.electricalWork || {}
   529→  if (ew.lightType && ew.quantity) {
   530→    const qty = parseFloat(ew.quantity) || 0
   531→    const rate = PRICES.electricalRates[ew.lightType as keyof typeof PRICES.electricalRates] || 0
   532→    if (qty > 0 && rate > 0) {
   533→      items.push({
   534→        label: 'Electrical Work',
   535→        subLabel: ew.lightType,
   536→        quantity: qty,
   537→        totalSqft: qty,
   538→        rate,
   539→        amount: qty * rate
   540→      })
   541→    }
   542→  }
   543→
   544→  // Painting
   545→  const pt = miscEstimate.painting || {}
   546→  if (pt.paintType && pt.totalArea) {
   547→    const area = parseFloat(pt.totalArea) || 0
   548→    const rate = PRICES.paintRates[pt.paintType as keyof typeof PRICES.paintRates] || 0
   549→    if (area > 0 && rate > 0) {
   550→      items.push({
   551→        label: 'Painting',
   552→        subLabel: pt.paintType,
   553→        sqft: area,
   554→        quantity: 1,
   555→        totalSqft: area,
   556→        rate,
   557→        amount: area * rate
   558→      })
   559→    }
   560→  }
   561→

   565→function getBedroomItems(bedroomEstimate: any): ExportItem[] {
   566→  const items: ExportItem[] = []
   567→  if (!bedroomEstimate) return items
   568→
   569→  // Wardrobe
   570→  const w = bedroomEstimate.wardrobe || {}
   571→  if (w.height && w.width && w.finish && w.wardrobeType) {
   572→    const sqft = (parseFloat(w.height) * parseFloat(w.width)) / 92903
   573→    const rate = PRICES.wardrobeFinish[w.finish as keyof typeof PRICES.wardrobeFinish] || 0
   574→    if (sqft > 0 && rate > 0) {
   575→      let amount = sqft * rate
   576→      const subLabel = `${w.wardrobeType} - ${w.finish}`
   577→      if (w.wardrobeType === 'Sliding' && w.slidingMechanism) {
   578→        amount += PRICES.wardrobeSlidingMechanism
   579→        items.push({
   580→          label: 'Wardrobe',
   581→          subLabel,
   582→          l: parseFloat(w.height) || 0,
   583→          b: parseFloat(w.width) || 0,
   584→          sqft,
   585→          quantity: 1,
   586→          totalSqft: sqft,
   587→          rate,
   588→          amount: sqft * rate
   589→        })
   590→        items.push({
   591→          label: '',
   592→          subLabel: 'Sliding Mechanism (Add-on)',
   593→          quantity: 1,
   594→          totalSqft: 1,
   595→          rate: PRICES.wardrobeSlidingMechanism,
   596→          amount: PRICES.wardrobeSlidingMechanism
   597→        })
   598→      } else {
   599→        items.push({
   600→          label: 'Wardrobe',
   601→          subLabel,
   602→          l: parseFloat(w.height) || 0,
   603→          b: parseFloat(w.width) || 0,
   604→          sqft,
   605→          quantity: 1,
   606→          totalSqft: sqft,
   607→          rate,
   608→          amount
   609→        })
   610→      }
   611→    }
   612→  }
   613→
   614→  // Bedroom Loft
   615→  const l = bedroomEstimate.loft || {}
   616→  if (l.height && l.width && l.loftType && l.finish) {
   617→    const sqft = (parseFloat(l.height) * parseFloat(l.width)) / 92903
   618→    const loftType = l.loftType as keyof typeof PRICES.bedroomLoftFinish
   619→    const finish = l.finish as 'SF' | 'HGL' | 'Acrylic'
   620→    if (sqft > 0 && loftType && finish && PRICES.bedroomLoftFinish[loftType]) {
   621→      const rate = PRICES.bedroomLoftFinish[loftType][finish] || 0
   622→      if (rate > 0) {
   623→        items.push({
   624→          label: 'Loft',
   625→          subLabel: `${l.loftType} - ${l.finish}`,
   626→          l: parseFloat(l.height) || 0,
   627→          b: parseFloat(l.width) || 0,
   628→          sqft,
   629→          quantity: 1,
   630→          totalSqft: sqft,
   631→          rate,
   632→          amount: sqft * rate
   633→        })
   634→      }
   635→    }
   636→  }
   637→
   638→  // Bed
   639→  const bd = bedroomEstimate.bed || {}
   640→  if (bd.length && bd.width && bd.finish) {
   641→    const sqft = (parseFloat(bd.length) * parseFloat(bd.width)) / 92903
   642→    const rate = PRICES.bedFinish[bd.finish as keyof typeof PRICES.bedFinish] || 0
   643→    if (sqft > 0 && rate > 0) {
   644→      const subLabel = `${bd.bedSize || ''} ${bd.storageType || ''} ${bd.finish}`.trim()
   645→      items.push({
   646→        label: 'Bed',
   647→        subLabel,
   648→        l: parseFloat(bd.length) || 0,
   649→        b: parseFloat(bd.width) || 0,
   650→        sqft,
   651→        quantity: 1,
   652→        totalSqft: sqft,
   653→        rate,
   654→        amount: sqft * rate
   655→      })
   656→    }
   657→  }
   658→
   659→  // Head Board
   660→  const hb = bedroomEstimate.headBoard || {}
   661→  if (hb.length && hb.width && hb.headBoardType) {
   662→    const sqft = (parseFloat(hb.length) * parseFloat(hb.width)) / 92903
   663→    const rate = PRICES.headBoardRates[hb.headBoardType as keyof typeof PRICES.headBoardRates] || 0
   664→    if (sqft > 0 && rate > 0) {
   665→      items.push({
   666→        label: 'Head Board',
   667→        subLabel: hb.headBoardType,
   668→        l: parseFloat(hb.length) || 0,
   669→        b: parseFloat(hb.width) || 0,
   670→        sqft,
   671→        quantity: 1,
   672→        totalSqft: sqft,
   673→        rate,
   674→        amount: sqft * rate
   675→      })
   676→    }
   677→  }
   678→
   679→  return items
   680→}

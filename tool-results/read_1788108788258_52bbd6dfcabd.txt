    89→const calculateSqft = (height: string, width: string): number => {
    90→  const h = parseFloat(height) || 0
    91→  const w = parseFloat(width) || 0
    92→  return (h * w) / 92903

   107→function getKitchenItems(components: any, kitchenType: string): ExportItem[] {

   110→  // Component 1: Ply Verticals or Structure/Countertop
   111→  const c1 = components.component1 || {}

   113→    const qty = parseFloat(c1.quantity) || 0
   114→    if (qty > 0) {
   115→      items.push({
   116→        label: 'Ply Verticals',
   117→        subLabel: 'CARCASE',
   118→        quantity: qty,
   119→        totalSqft: qty,
   120→        rate: PRICES.plyVerticals,
   121→        amount: qty * PRICES.plyVerticals
   122→      })
   123→    }
   124→  } else {
   125→    const sqft = calculateSqft(c1.height, c1.width)
   126→    const basePrice = PRICES.countertopMaterial[c1.material as keyof typeof PRICES.countertopMaterial] || PRICES.countertopMaterial.Granite

   128→      items.push({
   129→        label: 'Structure / Countertop',
   130→        subLabel: c1.material || 'Granite',
   131→        l: parseFloat(c1.height) || 0,
   132→        b: parseFloat(c1.width) || 0,
   133→        sqft,
   134→        quantity: 1,

   136→        rate: basePrice,
   137→        amount: sqft * basePrice
   138→      })
   139→    }
   140→  }
   141→
   142→  // Tandem Drawers

   148→      items.push({
   149→        label: 'Tandem Drawers',
   150→        subLabel: td.brand,

   338→    if (ol.loftType === 'Box Loft') {
   339→      const depth = (ol.depth || '450mm') as '450mm' | '600mm'
   340→      if (ol.finish === 'Postforming') {

   416→      rate: handlePrice,
   417→      amount: handleFeet * handlePrice
   418→    })
   419→  }
   420→

   556→        rate,
   557→        amount: sqft * rate
   558→      })
   559→    }
   560→  }
   561→
   562→  // Sitting with Cushion
   563→  const swc = comps.sittingWithCushion || {}
   564→  const swcSqft = calculateSqft(swc.height, swc.width)

   574→      rate: PRICES.sittingWithCushion,
   575→      amount: swcSqft * PRICES.sittingWithCushion
   576→    })
   577→  }
   578→
   579→  return items

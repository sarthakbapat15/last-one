   511→
   512→      case 'baseCarcase': {
   513→        const baseCarcaseSqft = calculateSqft(comp.height, comp.width)
   514→        return baseCarcaseSqft * prices.baseCarcase
   515→      }
   516→
   517→      case 'kitchenPaneling': {
   518→        const panelingSqft = calculateSqft(comp.height, comp.width)
   519→        const panelingFinish = comp.tallPantryFinish
   520→        const panelingRate = getEffectiveRate(panelingFinish, prices.kitchenPaneling)
   521→        return panelingSqft * panelingRate
   522→      }
   523→
   524→      case 'overheadCabinet': {
   525→        const overheadCabinetSqft = calculateSqft(comp.height, comp.width)
   526→        const ohcRate = getEffectiveRate(comp.overheadCabinetFinish, prices.overheadCabinetFinish)
   527→        return overheadCabinetSqft * ohcRate
   528→      }
   529→
   530→      case 'overheadLoft': {
   531→        const loftSqft = calculateSqft(comp.height, comp.width)
   532→        const loftType = comp.loftType as LoftType
   533→        const finish = comp.finish
   534→        if (loftType === 'Box Loft') {
   535→          const depth = (comp.depth || '450mm') as TallUnitDepth
   536→          if (finish === 'Postforming') return loftSqft * (parseFloat(postformingRate) || 0)
   537→          const boxPrices = prices.overheadBoxLoftFinish[depth]
   538→          const rate = boxPrices?.[finish as BoxLoftFinishType] || 0
   539→          return loftSqft * rate
   540→        }
   541→        if (loftType && prices.overheadLoft[loftType]) {
   542→          const basePrice = prices.overheadLoft[loftType]
   543→          const finishPrice = getEffectiveRate(finish, prices.overheadFinish)
   544→          return loftSqft * (basePrice + finishPrice)
   545→        }
   546→        return 0
   547→      }
   548→
   549→      case 'profileShutter': {
   550→        const sqft = calculateSqft(comp.height, comp.width)
   551→        const glassType = comp.tallPantryFinish as ProfileShutterGlassType
   552→        const rate = glassType ? (prices.profileShutterGlass[glassType] || 0) : 0
   553→        return sqft * rate
   554→      }
   555→
   556→      case 'magicCorner': {
   557→        const mcType = comp.brand as MagicCornerType
   558→        if (mcType && prices.magicCorner[mcType]) {
   559→          return prices.magicCorner[mcType]
   560→        }
   561→        return 0
   562→      }
   563→
   564→      case 'rollingShutter': {
   565→        const rsType = comp.brand as RollingShutterType
   566→        if (rsType && prices.rollingShutter[rsType]) {
   567→          return prices.rollingShutter[rsType]
   568→        }
   569→        return 0
   570→      }
   571→
   572→      case 'handles': {
   573→        const handleFeet = parseFloat(comp.runningFeet) || 0
   574→        const handlePrice = parseFloat(comp.handlePrice) || 0
   575→        return handleFeet * handlePrice
   576→      }
   577→
   578→      default:
   579→        return 0
   580→    }
   581→  }

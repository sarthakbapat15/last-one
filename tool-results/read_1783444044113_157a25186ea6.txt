   510→    const width = parseFloat(w) || 0
   511→    return (height * width) / 92903
   512→  }

   515→    const sqft = calcBRSqft(br.wardrobe.height, br.wardrobe.width)
   516→    const rate = getEffectiveRate(br.wardrobe.finish, prices.bedroomWardrobeFinish)
   517→    let total = sqft * rate
   518→    if (br.wardrobe.slidingMechanism) total += prices.bedroomWardrobeSlidingMechanism
   519→    return total
   520→  }
   521→
   522→  const calculateBedroomLoftTotal = (br: BedroomData): number => {
   523→    const sqft = calcBRSqft(br.loft.height, br.loft.width)
   524→    if (br.loft.finish === 'Postforming') {
   525→      return sqft * (parseFloat(postformingRate) || 0)
   526→    }
   527→    const rate = br.loft.loftType && br.loft.finish
   528→      ? (prices.bedroomLoftFinish[br.loft.loftType]?.[br.loft.finish] || 0)
   529→      : 0
   530→    return sqft * rate
   531→  }
   532→
   533→  const calculateWindowSeatTotal = (br: BedroomData): number => {
   534→    const sqft = calcBRSqft(br.windowSeat.height, br.windowSeat.width)
   535→    return sqft * getEffectiveRate(br.windowSeat.finish, prices.bedroomTallUnitFinish)
   536→  }

   539→    const baseSqft = calcBRSqft(br.studyTable.base.height, br.studyTable.base.width)
   540→    const ohSqft = calcBRSqft(br.studyTable.overhead.height, br.studyTable.overhead.width)
   541→    return (baseSqft + ohSqft) * getEffectiveRate(br.studyTable.finish, prices.bedroomTallUnitFinish)
   542→  }
   543→
   544→  const calculateDresserUnitTotal = (br: BedroomData): number => {
   545→    const baseSqft = calcBRSqft(br.dresserUnit.baseDrawers.height, br.dresserUnit.baseDrawers.width)
   546→    const msSqft = calcBRSqft(br.dresserUnit.mirrorWithStorage.height, br.dresserUnit.mirrorWithStorage.width)
   547→    const mbpSqft = calcBRSqft(br.dresserUnit.mirrorOnBackPanel.height, br.dresserUnit.mirrorOnBackPanel.width)
   548→    return (baseSqft + msSqft + mbpSqft) * getEffectiveRate(br.dresserUnit.finish, prices.bedroomTallUnitFinish)
   549→  }

   551→  const calculateBedTotal = (br: BedroomData): number => {
   552→    if (br.bed.typeOfBed === 'Open Bed with Legs') return prices.bedroomOpenBedPrice
   553→    const sqft = calcBRSqft(br.bed.height, br.bed.width)
   554→    const rate = getEffectiveRate(br.bed.finish, prices.bedroomTallUnitFinish)
   555→    let total = sqft * rate
   556→    if (br.bed.typeOfBed === 'Hydraulic (Automatic)' || br.bed.typeOfBed === 'Pullout Trolly Bed') {
   557→      total += prices.bedroomHydraulicMechanismPrice
   558→    }
   559→    return total
   560→  }

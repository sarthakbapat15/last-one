    65→const calculateSqft = (height: string, width: string): number => {
    66→  const h = parseFloat(height) || 0
    67→  const w = parseFloat(width) || 0
    68→  return (h * w) / 92903

    83→function getKitchenItems(components: any, kitchenType: string): ExportItem[] {

    86→  // Component 1: Ply Verticals or Structure/Countertop
    87→  const c1 = components.component1 || {}

    89→    const qty = parseFloat(c1.quantity) || 0
    90→    if (qty > 0) {
    91→      items.push({
    92→        label: 'Ply Verticals',
    93→        subLabel: 'CARCASE',
    94→        quantity: qty,
    95→        totalSqft: qty,

    97→        amount: qty * PRICES.plyVerticals

   100→  } else {
   101→    const sqft = calculateSqft(c1.height, c1.width)
   102→    const basePrice = c1.material === 'Quartz' ? 2000 : 1500

   104→      items.push({
   105→        label: 'Structure / Countertop',
   106→        subLabel: c1.material || 'Granite',

   110→        quantity: 1,

   112→        rate: basePrice,
   113→        amount: sqft * basePrice
   114→      })
   115→    }
   116→  }
   117→
   118→  // Tandem Drawers
   119→  const td = components.tandemDrawers || {}

   283→    const basePrice = PRICES.overheadLoft[ol.loftType as keyof typeof PRICES.overheadLoft] || 0
   284→    const finishPrice = ol.finish ? (PRICES.overheadFinish[ol.finish as keyof typeof PRICES.overheadFinish] || 0) : 0
   285→    const totalRate = basePrice + finishPrice
   286→    if (sqft > 0 && totalRate > 0) {

   491→function getMiscItems(miscEstimate: any): ExportItem[] {
   492→  const items: ExportItem[] = []
   493→  if (!miscEstimate) return items
   494→
   495→  // False Ceiling
   496→  const fc = miscEstimate.falseCeiling || {}
   497→  if (fc.height && fc.width && fc.material) {
   498→    const sqft = (parseFloat(fc.height) * parseFloat(fc.width)) / 92903
   499→    const rate = PRICES.falseCeilingMaterial[fc.material as keyof typeof PRICES.falseCeilingMaterial] || 0
   500→    if (sqft > 0 && rate > 0) {
   501→      items.push({
   502→        label: 'False Ceiling',
   503→        subLabel: `${fc.ceilingType || ''} - ${fc.material}`,
   504→        l: parseFloat(fc.height) || 0,
   505→        b: parseFloat(fc.width) || 0,
   506→        sqft,
   507→        quantity: 1,
   508→        totalSqft: sqft,
   509→        rate,
   510→        amount: sqft * rate
   511→      })
   512→    }
   513→  }
   514→
   515→  // Electrical Work
   516→  const ew = miscEstimate.electricalWork || {}
   517→  if (ew.lightType && ew.quantity) {
   518→    const qty = parseFloat(ew.quantity) || 0
   519→    const rate = PRICES.electricalRates[ew.lightType as keyof typeof PRICES.electricalRates] || 0
   520→    if (qty > 0 && rate > 0) {
   521→      items.push({
   522→        label: 'Electrical Work',
   523→        subLabel: ew.lightType,
   524→        quantity: qty,
   525→        totalSqft: qty,
   526→        rate,
   527→        amount: qty * rate
   528→      })
   529→    }
   530→  }
   531→
   532→  // Painting
   533→  const pt = miscEstimate.painting || {}
   534→  if (pt.paintType && pt.totalArea) {
   535→    const area = parseFloat(pt.totalArea) || 0
   536→    const rate = PRICES.paintRates[pt.paintType as keyof typeof PRICES.paintRates] || 0
   537→    if (area > 0 && rate > 0) {
   538→      items.push({
   539→        label: 'Painting',
   540→        subLabel: pt.paintType,
   541→        sqft: area,
   542→        quantity: 1,
   543→        totalSqft: area,
   544→        rate,
   545→        amount: area * rate
   546→      })
   547→    }
   548→  }
   549→
   550→  return items
   551→}

   553→export async function POST(request: NextRequest) {
   554→  try {
   555→    const body = await request.json()
   556→    const validatedData = validateData(body)
   557→
   558→    const workbook = new Workbook()
   559→
   560→    // ========================================
   561→    // SHEET 1: Quotation (olive+HGL format)
   562→    // ========================================
   563→    const qs = workbook.addWorksheet('Quotation')
   564→
   565→    // Company Header
   566→    qs.mergeCells('A1:H1')

   175→    switch (component) {
   176→      case 'component1':

   179→          return qty * PRICES.plyVerticals

   181→          const sqft = calculateSqft(comp.height, comp.width)
   182→          const basePrice = comp.material === 'Quartz' ? 2000 : 1500
   183→          return sqft * basePrice
   184→        }
   185→
   186→      case 'tandemDrawers': {
   187→        const brand = comp.brand as Brand
   188→        if (brand && PRICES.tandemDrawers[brand]) {
   189→          const qty = parseFloat(comp.quantity) || 0
   190→          return qty * PRICES.tandemDrawers[brand]
   191→        }
   192→        return 0
   193→      }
   194→
   195→      case 'dustbinBTD': {

   198→          const qty = parseFloat(comp.quantity) || 0
   199→          return qty * PRICES.dustbinBTD[dustbinBrand]
   200→        }
   201→        return 0
   202→      }
   203→
   204→      case 'bottlePullout': {
   205→        const bottleBrand = comp.brand as Brand
   206→        if (bottleBrand && PRICES.bottlePullout[bottleBrand]) {

   209→        }
   210→        return 0
   211→      }
   212→
   213→      case 'wickerBasket': {
   214→        const wickerBrand = comp.brand as Brand
   215→        if (wickerBrand && PRICES.wickerBasket[wickerBrand]) {
   216→          const qty = parseFloat(comp.quantity) || 0
   217→          return qty * PRICES.wickerBasket[wickerBrand]
   218→        }
   219→        return 0
   220→      }
   221→

   223→        const tallSqft = calculateSqft(comp.height, comp.width)
   224→        const tallFinish = comp.tallPantryFinish as TallPantryFinishType
   225→        if (tallFinish && PRICES.tallPantryFinish[tallFinish]) {
   226→          return tallSqft * PRICES.tallPantryFinish[tallFinish]
   227→        }
   228→        return 0
   229→      }
   230→
   231→      case 'pantryUnit': {

   233→        const pantryFinish = comp.tallPantryFinish as TallPantryFinishType

   236→          pantryTotal = pantrySqft * PRICES.tallPantryFinish[pantryFinish]
   237→        }
   238→        const accessories = comp.accessories as AccessoriesType
   239→        if (accessories && PRICES.pantryAccessories[accessories]) {
   240→          pantryTotal += PRICES.pantryAccessories[accessories]
   241→        }
   242→        return pantryTotal
   243→      }
   244→
   245→      case 'baseCarcase': {
   246→        const baseCarcaseSqft = calculateSqft(comp.height, comp.width)

   248→      }
   249→
   250→      case 'overheadCabinet': {
   251→        const overheadCabinetSqft = calculateSqft(comp.height, comp.width)
   252→        const ohcFinish = comp.overheadCabinetFinish as OverheadCabinetFinishType

    75→const calculateSqft = (height: string, width: string): number => {

   514→
   515→    // Wardrobe

   517→    if (wardrobe.height && wardrobe.width && wardrobe.finish) {
   518→      const sqft = calculateSqft(wardrobe.height, wardrobe.width)
   519→      const rate = PRICES.wardrobeFinish[wardrobe.finish as keyof typeof PRICES.wardrobeFinish] || 0

   522→          label: `${catLabel} - Wardrobe`,

   525→          b: parseFloat(wardrobe.width) || 0,
   526→          sqft,
   527→          quantity: 1,
   528→          totalSqft: sqft,
   529→          rate,
   530→          amount: sqft * rate
   531→        })
   532→      }
   533→
   534→      // Sliding mechanism add-on
   535→      if (wardrobe.hasSlidingMechanism) {
   536→        items.push({
   537→          label: `${catLabel} - Wardrobe Sliding Mechananism`,
   538→          subLabel: 'Add-on',
   539→          quantity: 1,
   540→          totalSqft: 1,
   541→          rate: PRICES.wardrobeSlidingMechanism,
   542→          amount: PRICES.wardrobeSlidingMechanism
   543→        })
   544→      }
   545→    }
   546→
   547→    // Loft
   548→    const loft = catData.loft || {}
   549→    if (loft.height && loft.width && loft.loftType && loft.finish) {
   550→      const sqft = calculateSqft(loft.height, loft.width)

   612→function getMiscItems(miscEstimate: any): ExportItem[] {
   613→  const items: ExportItem[] = []
   614→  if (!miscEstimate) return items
   615→
   616→  // False Ceiling
   617→  const falseCeiling = miscEstimate.falseCeiling || {}
   618→  if (falseCeiling.height && falseCeiling.width && falseCeiling.material && falseCeiling.ceilingType) {
   619→    const sqft = calculateSqft(falseCeiling.height, falseCeiling.width)
   620→    const materialRate = PRICES.falseCeilingMaterial[falseCeiling.material as keyof typeof PRICES.falseCeilingMaterial] || 0
   621→    const multiplier = PRICES.falseCeilingMultiplier[falseCeiling.ceilingType as keyof typeof PRICES.falseCeilingMultiplier] || 1
   622→    const totalRate = materialRate * multiplier
   623→    if (sqft > 0 && totalRate > 0) {
   624→      items.push({
   625→        label: 'False Ceiling',
   626→        subLabel: `${falseCeiling.material} / ${falseCeiling.ceilingType}`,
   627→        l: parseFloat(falseCeiling.height) || 0,
   628→        b: parseFloat(falseCeiling.width) || 0,
   629→        sqft,
   630→        quantity: 1,
   631→        totalSqft: sqft,
   632→        rate: totalRate,
   633→        amount: sqft * totalRate
   634→      })
   635→    }
   636→  }
   637→
   638→  // Electrical
   639→  const electrical = miscEstimate.electrical || {}
   640→  if (electrical.items && Array.isArray(electrical.items)) {
   641→    electrical.items.forEach((item: any) => {
   642→      const qty = parseFloat(item.quantity) || 0
   643→      const rate = PRICES.electricalRates[item.type as keyof typeof PRICES.electricalRates] || 0
   644→      if (qty > 0 && rate > 0) {
   645→        items.push({
   646→          label: 'Electrical',
   647→          subLabel: item.type,
   648→          quantity: qty,
   649→          totalSqft: qty,
   650→          rate,
   651→          amount: qty * rate
   652→        })
   653→      }
   654→    })
   655→  }
   656→
   657→  // Painting
   658→  const painting = miscEstimate.painting || {}
   659→  if (painting.items && Array.isArray(painting.items)) {
   660→    painting.items.forEach((item: any) => {
   661→      const area = parseFloat(item.area) || 0
   662→      const rate = PRICES.paintRates[item.type as keyof typeof PRICES.paintRates] || 0
   663→      if (area > 0 && rate > 0) {
   664→        items.push({
   665→          label: 'Painting',
   666→          subLabel: item.type,
   667→          quantity: area,
   668→          totalSqft: area,
   669→          rate,
   670→          amount: area * rate
   671→        })
   672→      }
   673→    })
   674→  }
   675→
   676→  return items
   677→}
